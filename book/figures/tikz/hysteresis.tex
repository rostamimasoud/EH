% A fold bifurcation with hysteresis, drawn generically.
%
% Deliberately carries no cavity specific numbers. The source study computes
% these curves for eleven real cavities; this diagram exists only so that a
% reader who has not met a fold before can see what the words mean.
\begin{tikzpicture}[
  font=\footnotesize,
  note/.style={font=\scriptsize, text=black!62, align=left, inner sep=0pt},
  lbl/.style={font=\scriptsize\sffamily\bfseries, inner sep=0pt},
  arr/.style={-{Stealth[length=4pt,width=3pt]}, line width=0.8pt},
]

\definecolor{ccold}{HTML}{2E6F95}
\definecolor{cwarm}{HTML}{A8402F}
\definecolor{ink}{HTML}{3A3A3A}

% axes
\draw[ink!70, line width=0.6pt, -{Stealth[length=4pt]}] (0,0) -- (9.4,0);
\draw[ink!70, line width=0.6pt, -{Stealth[length=4pt]}] (0,0) -- (0,4.6);
\node[note, anchor=north] at (4.7,-0.24) {sea ice produced in the polynya};
\node[note, anchor=south, rotate=90] at (-0.32,2.3) {basal melt rate};

% shaded bistable window
\fill[ink!7] (2.6,0.05) rectangle (7.0,4.3);
\node[note, anchor=south, text=ink] at (4.8,4.34) {both states possible};

% fold positions
\draw[ink!40, line width=0.5pt, dashed] (2.6,0) -- (2.6,4.3);
\draw[ink!40, line width=0.5pt, dashed] (7.0,0) -- (7.0,4.3);

% warm (high melt) stable branch: exists at low forcing, ends at right fold
\draw[cwarm, line width=1.5pt]
  (0.55,3.95) .. controls (2.6,3.75) and (4.8,3.55) .. (7.0,3.05);
\node[lbl, text=cwarm, anchor=west] at (0.62,4.18) {warm state};

% cold (low melt) stable branch: exists at high forcing, ends at left fold
\draw[ccold, line width=1.5pt]
  (2.6,0.60) .. controls (4.8,0.42) and (6.8,0.36) .. (9.1,0.32);
\node[lbl, text=ccold, anchor=east] at (9.05,0.62) {cold state};

% unstable middle branch
\draw[ink!55, line width=1.0pt, dashed]
  (2.6,0.60) .. controls (4.4,1.55) and (5.4,2.35) .. (7.0,3.05);
\node[note, text=ink!60, anchor=west, text width=2.0cm] at (4.05,2.05)
  {unstable: no real cavity sits here};

% fold markers
\fill[ink] (2.6,0.60) circle (0.085);
\fill[ink] (7.0,3.05) circle (0.085);
\node[note, anchor=north east, text width=1.7cm] at (2.5,0.48)
  {lower fold: the cold state ceases to exist};
\node[note, anchor=south west, text width=1.7cm] at (7.12,3.15)
  {upper fold: the warm state ceases to exist};

% transition arrows
\draw[arr, cwarm] (2.6,0.78) -- (2.6,3.72);
\node[note, text=cwarm, anchor=east, text width=1.35cm] at (2.48,2.35)
  {tips abruptly};
\draw[arr, ccold] (7.0,2.88) -- (7.0,0.52);
\node[note, text=ccold!80!black, anchor=west, text width=1.5cm] at (7.12,1.70)
  {recovers only here};

\end{tikzpicture}
