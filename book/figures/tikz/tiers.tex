% The tiered construction of the composite score: three levels of spatial claim.
% Drawn for this book.
\begin{tikzpicture}[
  font=\footnotesize,
  box/.style={draw=#1, fill=#1!10, line width=0.7pt, rounded corners=1.5pt,
              minimum height=0.95cm, align=center, inner sep=4pt},
  ttl/.style={font=\scriptsize\sffamily\bfseries},
  note/.style={font=\scriptsize, text=black!62, align=left},
  flow/.style={-{Stealth[length=4pt,width=3pt]}, line width=0.7pt, black!55},
]

\definecolor{t1}{HTML}{2E6F95}
\definecolor{t2}{HTML}{B07A2B}
\definecolor{t3}{HTML}{1B7A6E}
\definecolor{ink}{HTML}{3A3A3A}

% ---- tier i ---------------------------------------------------------------
\node[box=t1, text width=3.0cm] (a) at (0,0)
  {surface warming\\[1pt] \scriptsize carries a real spatial pattern};
\node[ttl, text=t1, anchor=south west] at (-1.55,0.56) {tier one};
\node[note, anchor=west, text width=4.2cm] at (1.75,0)
  {latitude and land modulate the global signal, so the field varies smoothly
   from place to place};

% a small latitudinal band sketch
\begin{scope}[xshift=-1.5cm, yshift=-1.55cm]
  \foreach \i/\c in {0/20,1/38,2/58,3/38,4/20}{
    \fill[t1!\c] (\i*0.24,0) rectangle (\i*0.24+0.24,0.42);
  }
  \draw[ink!40, line width=0.3pt] (0,0) rectangle (1.2,0.42);
\end{scope}

% ---- tier ii --------------------------------------------------------------
\node[box=t2, text width=3.0cm] (b) at (0,-2.45)
  {water hazard field\\[1pt] \scriptsize carries observed geography};
\node[ttl, text=t2, anchor=south west] at (-1.55,-1.89) {tier two};
\node[note, anchor=west, text width=4.2cm] at (1.75,-2.45)
  {a real gridded field, irreducible to smooth functions, supplying the hotspots
   that make the map recognisable};

\begin{scope}[xshift=-1.5cm, yshift=-4.0cm]
  \foreach \i/\j/\c in {0/0/15, 1/0/60, 2/0/25, 3/0/70, 4/0/10,
                        0/1/45, 1/1/12, 2/1/80, 3/1/20, 4/1/35}{
    \fill[t2!\c] (\i*0.24,\j*0.21) rectangle (\i*0.24+0.24,\j*0.21+0.21);
  }
  \draw[ink!40, line width=0.3pt] (0,0) rectangle (1.2,0.42);
\end{scope}

% ---- tier iii -------------------------------------------------------------
\node[box=t3, text width=3.0cm] (c) at (0,-4.9)
  {ocean and carbon\\[1pt] \scriptsize no spatial claim made};
\node[ttl, text=t3, anchor=south west] at (-1.55,-4.34) {tier three};
\node[note, anchor=west, text width=4.2cm] at (1.75,-4.9)
  {global quantities enter every land cell identically, because the model has no
   honest basis for downscaling them};

\begin{scope}[xshift=-1.5cm, yshift=-6.45cm]
  \fill[t3!35] (0,0) rectangle (1.2,0.42);
  \draw[ink!40, line width=0.3pt] (0,0) rectangle (1.2,0.42);
\end{scope}

% ---- combine --------------------------------------------------------------
\node[box=ink, text width=3.0cm, fill=ink!7] (s) at (0,-7.55)
  {composite score\\[1pt] \scriptsize one value per land cell};

\draw[flow] (a.south) -- (b.north);
\draw[flow] (b.south) -- (c.north);
\draw[flow] (c.south) -- (s.north);

\node[note, anchor=west, text width=4.2cm] at (1.75,-7.55)
  {each variable enters at the level of spatial detail it can actually support,
   so the map never claims resolution it has not earned};

\end{tikzpicture}
