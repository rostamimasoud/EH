% Recoverable against unrecoverable compaction, and the delay that hides it.
%
% Two panels on a strict grid. Each panel occupies a 5.4 by 3.2 box, axis labels
% sit outside that box, and the panel caption sits in a reserved band below
% everything, so no two text blocks can meet.
\begin{tikzpicture}[
  font=\footnotesize,
  note/.style={font=\scriptsize, text=black!62, align=left, inner sep=0pt},
  ttl/.style={font=\scriptsize\sffamily\bfseries, inner sep=0pt},
  pcap/.style={font=\scriptsize, text=black!75, align=left, inner sep=0pt},
]

\definecolor{cel}{HTML}{2E6F95}
\definecolor{cin}{HTML}{8C4A32}
\definecolor{ink}{HTML}{3A3A3A}

% =========================================================== panel a
\begin{scope}
  % axes
  \draw[ink!70, line width=0.6pt, -{Stealth[length=4pt]}] (0,0) -- (5.5,0);
  \draw[ink!70, line width=0.6pt, -{Stealth[length=4pt]}] (0,0) -- (0,3.30);
  \node[note, anchor=north] at (2.75,-0.22) {load carried by the sediment};
  \node[note, anchor=south, rotate=90] at (-0.30,1.65) {thickness lost};

  % preconsolidation marker, label above the plot area
  \draw[ink!45, line width=0.6pt, dashed] (2.55,0) -- (2.55,2.95);
  \node[ttl, text=ink, anchor=south] at (2.55,3.36)
    {greatest load ever carried};

  % elastic branch
  \draw[cel, line width=1.5pt] (0.30,0.18) -- (2.55,0.66);
  \draw[cel!70, line width=0.7pt, -{Stealth[length=3pt]}]
    (2.05,0.55) .. controls (1.60,0.34) .. (1.15,0.32);
  \node[note, text=cel!62!black, anchor=north west, text width=2.15cm]
    at (0.16,1.72) {below this load the clay springs back and nothing is lost};

  % inelastic branch
  \draw[cin, line width=1.7pt] (2.55,0.66) -- (5.20,2.85);
  \fill[cel] (2.55,0.66) circle (0.075);
  \node[note, text=cin, anchor=north west, text width=2.10cm]
    at (2.70,2.88) {beyond it the grains rearrange and stay rearranged};

  \node[pcap, anchor=north west, text width=5.5cm] at (0,-0.70)
    {(a) pumping transfers the weight above onto the sediment itself};
\end{scope}

% =========================================================== panel b
\begin{scope}[xshift=7.60cm]
  \draw[ink!70, line width=0.6pt, -{Stealth[length=4pt]}] (0,0) -- (5.5,0);
  \draw[ink!70, line width=0.6pt, -{Stealth[length=4pt]}] (0,0) -- (0,3.30);
  \node[note, anchor=north] at (2.75,-0.22) {time};
  \node[note, anchor=south, rotate=90] at (-0.30,1.65) {rate of loss};

  \draw[ink!45, line width=0.6pt, dashed] (2.30,0) -- (2.30,2.95);
  \node[ttl, text=ink, anchor=south] at (2.30,3.36) {all pumping stops};

  \fill[cin!13]
    (2.30,2.45) .. controls (2.85,2.32) and (3.30,1.70) .. (3.95,1.15)
    .. controls (4.50,0.78) and (4.95,0.62) .. (5.30,0.56)
    -- (5.30,0) -- (2.30,0) -- cycle;

  \draw[cin, line width=1.6pt]
    (0.30,0.38) .. controls (1.30,1.65) and (1.85,2.28) .. (2.30,2.45)
    .. controls (2.85,2.32) and (3.30,1.70) .. (3.95,1.15)
    .. controls (4.50,0.78) and (4.95,0.62) .. (5.30,0.56);

  \node[note, text=cin, anchor=north west, text width=1.90cm]
    at (3.62,3.02) {the commitment: loss that continues after the cause is
    removed};

  \node[pcap, anchor=north west, text width=5.5cm] at (0,-0.70)
    {(b) thick clay drains slowly, so the ground keeps sinking for decades};
\end{scope}

\end{tikzpicture}
